module alu(input A1, A0, B1, B0, Op1, Op0, output Y1, Y0);
    wire not_Op1, not_Op0;
    wire sel0, sel1, sel2, sel3;

    not n1(not_Op1, Op1);
    not n0(not_Op0, Op0);
    and a0(sel0, not_Op1, not_Op0);
    and a1(sel1, not_Op1, Op0);
    and a2(sel2, Op1, not_Op0);
    and a3(sel3, Op1, Op0);

    wire a0_add, a1_add, a0_sub, a1_sub;
    wire b0_not, b1_not, b0_sub, b1_sub;
    wire a0b0_and, a1b1_and, a0b0_or, a1b1_or;

    not n2(b0_not, B0);
    not n3(b1_not, B1);
    xor x0(b0_sub, B0, sel1);
    xor x1(b1_sub, B1, sel1);
    xor x2(a0_add, A0, b0_sub);
    xor x3(a1_add, A1, b1_sub);
    and a4(a0b0_and, A0, B0);
    and a5(a1b1_and, A1, B1);
    or o0(a0b0_or, A0, B0);
    or o1(a1b1_or, A1, B1);
    xor x4(a0_sub, A0, B0);
    xor x5(a1_sub, A1, B1);

    wire y0_0, y0_1, y0_2, y0_3;
    wire y1_0, y1_1, y1_2, y1_3;

    and a6(y0_0, sel0, a0_add);
    and a7(y0_1, sel1, a0_sub);
    and a8(y0_2, sel2, a0b0_and);
    and a9(y0_3, sel3, a0b0_or);
    or o2(Y0, y0_0, y0_1);
    wire y0_tmp;
    or o3(y0_tmp, y0_2, y0_3);
    or o4(Y0, Y0, y0_tmp);

    and a10(y1_0, sel0, a1_add);
    and a11(y1_1, sel1, a1_sub);
    and a12(y1_2, sel2, a1b1_and);
    and a13(y1_3, sel3, a1b1_or);
    or o5(Y1, y1_0, y1_1);
    wire y1_tmp;
    or o6(y1_tmp, y1_2, y1_3);
    or o7(Y1, Y1, y1_tmp);
endmodule
