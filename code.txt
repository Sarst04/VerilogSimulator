module FourBitAdder(
    input a3, a2, a1, a0, b3, b2, b1, b0,
    output s0, s1, s2, s3, c_out
);

    // Internal carry wires
    wire c1, c2, c3;

    // Bit 0 (LSB) - Half Adder
    xor #(2,1) XOR_0(s0, a0, b0);
    and #(2,1) AND_0(c1, a0, b0);

    // Bit 1 - Full Adder
    wire s1_xor1;
    xor #(2,1) XOR1_1(s1_xor1, a1, b1);
    xor #(2,1) XOR2_1(s1, s1_xor1, c1);
    
    wire c2_and1, c2_and2;
    and #(2,1) AND1_1(c2_and1, a1, b1);
    and #(2,1) AND2_1(c2_and2, s1_xor1, c1);
    or #(2,1) OR_1(c2, c2_and1, c2_and2);

    // Bit 2 - Full Adder
    wire s2_xor1;
    xor #(2,1) XOR1_2(s2_xor1, a2, b2);
    xor #(2,1) XOR2_2(s2, s2_xor1, c2);
    
    wire c3_and1, c3_and2;
    and #(2,1) AND1_2(c3_and1, a2, b2);
    and #(2,1) AND2_2(c3_and2, s2_xor1, c2);
    or #(2,1) OR_2(c3, c3_and1, c3_and2);

    // Bit 3 (MSB) - Full Adder
    wire s3_xor1;
    xor #(2,1) XOR1_3(s3_xor1, a3, b3);
    xor #(2,1) XOR2_3(s3, s3_xor1, c3);
    
    wire c_out_and1, c_out_and2;
    and #(2,1) AND1_3(c_out_and1, a3, b3);
    and #(2,1) AND2_3(c_out_and2, s3_xor1, c3);
    or #(2,1) OR_3(c_out, c_out_and1, c_out_and2);

endmodule
